## AP_MotorsBlimp класс

```cpp
class AP_MotorsBlimp : public AP_Motors {
public:
    AP_MotorsBlimp(uint16_t speed_hz = AP_MOTORS_SPEED_DEFAULT);

    // AP_Motors pure virtual implementations
    void init(motor_frame_class frame_class, motor_frame_type frame_type) override;
    void output() override;
    void output_min() override;
    void set_desired_spool_state(DesiredSpoolState spool) override;
    float get_throttle_hover() const override { return 0.0f; }
    uint32_t get_motor_mask() override;
    void set_frame_class_and_type(motor_frame_class, motor_frame_type) override {}

protected:
    void output_armed_stabilizing() override;
    void update_throttle_filter() override;
    const char* _get_frame_string() const override { return "BLIMP"; }
    void _output_test_seq(uint8_t motor_seq, int16_t pwm) override;

    // Mixing matrix factors
    float _roll_factor[4];
    float _pitch_factor[4];
    float _yaw_factor[4];
    float _throttle_factor[4];  // X axis (forward/backward)
};
```

## output_armed_stabilizing()

```cpp
void AP_MotorsBlimp::output_armed_stabilizing() {
    for (int i = 0; i < 4; i++) {
        float output = _roll_factor[i] * _roll_in
                     + _pitch_factor[i] * _pitch_in
                     + _yaw_factor[i] * _yaw_in
                     + _throttle_factor[i] * _throttle_in;
        output = constrain_float(output, 0.0f, 1.0f);
        rc_write(i, output_to_pwm(output));
    }
}
```

## Адаптация mode_manual

```cpp
void ModeManual::run() {
    Vector3f pilot;
    float pilot_yaw;
    get_pilot_input(pilot, pilot_yaw);

    // pilot.x = [-1,1] → throttle [0,1]
    float throttle = (pilot.x + 1.0f) * 0.5f;

    blimp.motors->set_roll(pilot.y);
    blimp.motors->set_pitch(pilot.z);
    blimp.motors->set_yaw(pilot_yaw);
    blimp.motors->set_throttle(throttle);
}
```
